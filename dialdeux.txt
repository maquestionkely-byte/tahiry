  const TARGET_FILE = 'dialle2.txt';

  // ==================== PARSEUR POUR EXPRESSIONS AVEC SIGNIFICATION ====================
  function parseExpressionsFile(rawText) {
    const lines = rawText.split(/\r?\n/);
    const rubriques = [];
    const rubriquesData = {};
    const rubriquesDescriptions = {};
    
    let currentTitre = null;
    let currentClasse = null;
    let currentSignification = null;
    let currentExpressions = [];
    let i = 0;
    
    function sauvegarderClasse() {
      if (currentTitre && currentClasse && currentExpressions.length > 0) {
        if (!rubriquesData[currentTitre]) {
          rubriquesData[currentTitre] = {};
        }
        rubriquesData[currentTitre][currentClasse.toLowerCase()] = {
          signification: currentSignification || "",
          expressions: currentExpressions.slice()
        };
        currentExpressions = [];
        currentSignification = null;
      }
    }
    
    while (i < lines.length) {
      let line = lines[i].trim();
      
      // Ignorer les lignes vides
      if (!line) {
        i++;
        continue;
      }
      
      // Détecter TITRE
      const titreMatch = line.match(/^[Tt]itre\s*[:;]\s*(.+)$/);
      if (titreMatch) {
        sauvegarderClasse();
        const nouveauTitre = titreMatch[1].trim();
        currentTitre = nouveauTitre;
        currentClasse = null;
        currentSignification = null;
        currentExpressions = [];
        
        if (!rubriques.includes(currentTitre)) {
          rubriques.push(currentTitre);
          rubriquesData[currentTitre] = {};
          rubriquesDescriptions[currentTitre] = "";
        }
        
        i++;
        continue;
      }
      
      // Détecter CLASSE (SOUTENU, COURANT, FAMILIER)
      const classeMatch = line.match(/^(SOUTENU|COURANT|FAMILIER)$/i);
      if (classeMatch && currentTitre) {
        sauvegarderClasse();
        currentClasse = classeMatch[1].toUpperCase();
        currentSignification = null;
        currentExpressions = [];
        i++;
        continue;
      }
      
      // Détecter SIGNIFICATION
      const sigMatch = line.match(/^signification\s*[:;]\s*(.+)$/i);
      if (sigMatch && currentTitre && currentClasse) {
        currentSignification = sigMatch[1].trim();
        i++;
        continue;
      }
      
      // Détecter une expression (commence par - ou •)
      const exprMatch = line.match(/^[-•]\s*(.+)$/);
      if (exprMatch && currentTitre && currentClasse) {
        currentExpressions.push(exprMatch[1].trim());
        i++;
        continue;
      }
      
      // Description (optionnel)
      const descMatch = line.match(/^[Dd]escription\s*[:;]\s*(.+)$/);
      if (descMatch && currentTitre) {
        rubriquesDescriptions[currentTitre] = descMatch[1].trim();
        i++;
        continue;
      }
      
      i++;
    }
    
    // Sauvegarder la dernière classe
    sauvegarderClasse();
    
    // Nettoyer les rubriques vides
    const rubriquesValides = rubriques.filter(r => {
      const data = rubriquesData[r];
      return data && (data.soutenu || data.courant || data.familier);
    });
    
    const rubriquesDataFiltre = {};
    const rubriquesDescriptionsFiltre = {};
    for (const r of rubriquesValides) {
      rubriquesDataFiltre[r] = rubriquesData[r];
      rubriquesDescriptionsFiltre[r] = rubriquesDescriptions[r] || "";
    }
    
    return {
      rubriques: rubriquesValides,
      rubriquesData: rubriquesDataFiltre,
      rubriquesDescriptions: rubriquesDescriptionsFiltre
    };
  }

  // ==================== INDEXEDDB ====================
  function openDB() {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(DB_NAME, 2);
      request.onupgradeneeded = (e) => {
        const db = e.target.result;
        if (!db.objectStoreNames.contains(STORE_NAME)) {
          db.createObjectStore(STORE_NAME);
        }
      };
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }
  
  async function saveToDB(content, version) {
    const db = await openDB();
    return new Promise((resolve, reject) => {
      const tx = db.transaction(STORE_NAME, 'readwrite');
      const store = tx.objectStore(STORE_NAME);
      store.put(content, FILE_KEY);
      store.put(version, VERSION_KEY);
      tx.oncomplete = () => resolve();
      tx.onerror = () => reject(tx.error);
    });
  }
  
  async function loadFromDB() {
    const db = await openDB();
    return new Promise((resolve) => {
      const tx = db.transaction(STORE_NAME, 'readonly');
      const store = tx.objectStore(STORE_NAME);
      const reqContent = store.get(FILE_KEY);
      const reqVersion = store.get(VERSION_KEY);
      let content = null, version = null;
      let completed = 0;
      function check() {
        completed++;
        if (completed === 2) resolve({ content, version });
      }
      reqContent.onsuccess = () => { content = reqContent.result; check(); };
      reqVersion.onsuccess = () => { version = reqVersion.result; check(); };
      reqContent.onerror = () => check();
      reqVersion.onerror = () => check();
    });
  }

  // ==================== VERSION DISTANTE ====================
  async function getRemoteVersionDate() {
    try {
      const res = await fetch(DATE_INFO_URL, { cache: 'no-cache' });
      if (!res.ok) return null;
      const text = await res.text();
      const blocks = text.split(/DERNIÈRE MISE À JOUR/);
      for (const block of blocks) {
        const fileMatch = block.match(/📁 Fichier:\s*([^\n]+)/);
        if (fileMatch && fileMatch[1].trim() === TARGET_FILE) {
          const dayMatch = block.match(/📅\s*([\d\/]+)/);
          const timeMatch = block.match(/⏰\s*([\d:]+)/);
          if (dayMatch && timeMatch) return `${dayMatch[1]} ${timeMatch[1]}`;
        }
      }
      return null;
    } catch (e) { return null; }
  }
  
  async function fetchRemoteTxt() {
    const res = await fetch(FILE_URL, { cache: 'no-store' });
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    return await res.text();
  }

  // ==================== MISE À JOUR SILENCIEUSE ====================
  async function silentUpdate() {
    if (isUpdating || !navigator.onLine) return false;
    try {
      const remoteVer = await getRemoteVersionDate();
      if (!remoteVer) return false;
      if (currentVersionDate === remoteVer) return false;
      
      isUpdating = true;
      const newContent = await fetchRemoteTxt();
      if (!newContent || newContent.length < 100) throw new Error("Contenu invalide");
      
      await saveToDB(newContent, remoteVer);
      currentVersionDate = remoteVer;
      
      const parsed = parseExpressionsFile(newContent);
      cachedRubriques = parsed.rubriques;
      cachedRubriquesData = parsed.rubriquesData;
      cachedDescriptions = parsed.rubriquesDescriptions;
      
      updateListOnly();
      
      isUpdating = false;
      showToast('✅ Expressions mises à jour');
      return true;
    } catch (e) {
      console.error(e);
      isUpdating = false;
      return false;
    }
  }

  function escapeHtml(str) {
    if (!str) return '';
    return str.replace(/[&<>]/g, m => m === '&' ? '&amp;' : m === '<' ? '&lt;' : '&gt;');
  }

  // ==================== RECHERCHE ====================
  function fuzzyMatch(text, search) {
    if (!search.trim()) return true;
    const t = text.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "");
    const s = search.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "");
    return t.includes(s);
  }

  function filterRubriques() {
    if (!currentSearch.trim()) return cachedRubriques;
    const term = currentSearch.trim();
    if (/^\d+$/.test(term)) {
      const num = parseInt(term);
      const found = cachedRubriques.find((_, idx) => idx + 1 === num);
      return found ? [found] : [];
    }
    return cachedRubriques.filter(t => fuzzyMatch(t, term));
  }

  // ============ MISE À JOUR UNIQUEMENT DE LA LISTE ============
  function updateListOnly() {
    if (!rubriquesGrid) return;
    const filtered = filterRubriques();
    
    if (searchBadgeSpan) {
      if (currentSearch.trim()) {
        searchBadgeSpan.innerHTML = `🔍 ${filtered.length} résultat(s) pour « ${escapeHtml(currentSearch)} »`;
      } else {
        searchBadgeSpan.innerHTML = ``;
      }
    }
    
    if (filtered.length === 0 && currentSearch.trim() !== "") {
      rubriquesGrid.innerHTML = `<div class="no-results">😢 Aucun thème ne correspond à « ${escapeHtml(currentSearch)} »</div>`;
      return;
    }
    
    const cards = filtered.map(titre => {
      const desc = cachedDescriptions[titre] || "";
      const num = cachedRubriques.findIndex(t => t === titre) + 1;
      return `
        <div class="rubric-card" data-titre="${escapeHtml(titre)}">
          <button class="rubric-btn">
            <div class="rubric-header">
              <span class="rubric-num-circle">${num}</span>
              <span class="rubric-title-text">${escapeHtml(titre)}</span>
              <span class="rubric-arrow">➡️</span>
            </div>
            ${desc ? `<div class="rubric-description">📝 ${escapeHtml(desc)}</div>` : ''}
          </button>
        </div>
      `;
    }).join('');
    
    rubriquesGrid.innerHTML = cards;
    
    document.querySelectorAll('.rubric-card').forEach(card => {
      card.addEventListener('click', () => {
        const titre = card.getAttribute('data-titre');
        if (titre) renderDialogue(titre);
      });
    });
  }

  async function initApp() {
    const app = document.getElementById('app-root');
    app.innerHTML = `<div class="loading-box"><div class="spinner"></div><p>Chargement des expressions...</p></div>`;
    
    try {
      const cached = await loadFromDB();
      
      if (cached.content && cached.content.trim().length > 50) {
        currentVersionDate = cached.version;
        const parsed = parseExpressionsFile(cached.content);
        cachedRubriques = parsed.rubriques;
        cachedRubriquesData = parsed.rubriquesData;
        cachedDescriptions = parsed.rubriquesDescriptions;
        
        if (cachedRubriques.length > 0) {
          renderList();
        } else {
          throw new Error("Fichier corrompu");
        }
      } else {
        if (!navigator.onLine) throw new Error("Pas de connexion internet");
        
        const fileContent = await fetchRemoteTxt();
        const remoteVersion = await getRemoteVersionDate();
        const finalVersion = remoteVersion || new Date().toLocaleString('fr-FR');
        
        await saveToDB(fileContent, finalVersion);
        currentVersionDate = finalVersion;
        
        const parsed = parseExpressionsFile(fileContent);
        cachedRubriques = parsed.rubriques;
        cachedRubriquesData = parsed.rubriquesData;
        cachedDescriptions = parsed.rubriquesDescriptions;
        
        if (cachedRubriques.length === 0) {
          throw new Error("Aucune expression trouvée");
        }
        
        renderList();
      }
      
      if (navigator.onLine) setTimeout(() => silentUpdate(), 2000);
      window.addEventListener('online', () => setTimeout(() => silentUpdate(), 1000));
      
    } catch (err) {
      console.error(err);
      // Afficher la page d'erreur au lieu du fallback
      renderErrorPage(err.message);
    }
  }
  
  initApp();
