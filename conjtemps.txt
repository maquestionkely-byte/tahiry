function showToast(msg, duration = 2000) {

function escapeHtml(str) {
  if (!str) return '';
  return str.replace(/[&<>]/g, m => m === '&' ? '&amp;' : m === '<' ? '&lt;' : '&gt;');
}

function fuzzyMatch(text, search) {
  if (!search.trim()) return true;
  const t = text.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "");
  const s = search.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "");
  return t.includes(s);
}

// ============================================================
// STOCKAGE — IndexedDB + fallback localStorage
// ============================================================
function openDB() {
  return new Promise((resolve, reject) => {
    let request;
    try {
      request = indexedDB.open(DB_NAME, 1);
    } catch (e) {
      return reject(e);
    }
    request.onupgradeneeded = (e) => {
      const db = e.target.result;
      if (!db.objectStoreNames.contains(STORE_NAME)) {
        db.createObjectStore(STORE_NAME);
      }
    };
    request.onsuccess = () => resolve(request.result);
    request.onerror  = () => reject(request.error);
    request.onblocked = () => reject(new Error("IndexedDB blocked"));
  });
}

function lsKey(key) { return 'fahaizana_' + key; }

function lsSave(key, content, version) {
  try {
    localStorage.setItem(lsKey(key), JSON.stringify({ content, version }));
    return true;
  } catch (e) {
    console.warn("localStorage save failed", e);
    return false;
  }
}

function lsLoad(key) {
  try {
    const raw = localStorage.getItem(lsKey(key));
    if (!raw) return null;
    return JSON.parse(raw);
  } catch (e) {
    return null;
  }
}

async function saveToDB(key, content, version) {
  if (useIndexedDB) {
    try {
      const db = await openDB();
      return new Promise((resolve, reject) => {
        const tx = db.transaction(STORE_NAME, 'readwrite');
        tx.objectStore(STORE_NAME).put({ content, version }, key);
        tx.oncomplete = () => resolve();
        tx.onerror    = () => reject(tx.error);
      });
    } catch (e) {
      console.warn("IndexedDB save failed, fallback localStorage:", e);
      useIndexedDB = false;
      lsSave(key, content, version);
    }
  } else {
    lsSave(key, content, version);
  }
}

async function loadFromDB(key) {
  if (useIndexedDB) {
    try {
      const db = await openDB();
      return await new Promise((resolve) => {
        const tx = db.transaction(STORE_NAME, 'readonly');
        const req = tx.objectStore(STORE_NAME).get(key);
        req.onsuccess = () => resolve(req.result || null);
        req.onerror   = () => resolve(null);
      });
    } catch (e) {
      console.warn("IndexedDB load failed, fallback localStorage:", e);
      useIndexedDB = false;
      return lsLoad(key);
    }
  } else {
    return lsLoad(key);
  }
}

// ============================================================
// VERSION DISTANTE
// ============================================================
async function getRemoteVersion(fichier) {
  try {
    const res = await fetch(VERSION_URL + '?t=' + Date.now(), { cache: 'no-cache' });
    if (!res.ok) return null;
    const text = await res.text();
    const blocks = text.split(/DERNIÈRE MISE À JOUR/);
    for (const block of blocks) {
      const fileMatch = block.match(/📁 Fichier:\s*([^\n]+)/);
      if (fileMatch && fileMatch[1].trim() === fichier) {
        const dayMatch  = block.match(/📅\s*([\d\/]+)/);
        const timeMatch = block.match(/⏰\s*([\d:]+)/);
        if (dayMatch && timeMatch) return `${dayMatch[1]} ${timeMatch[1]}`;
      }
    }
    return null;
  } catch (e) { return null; }
}

// ============================================================
// FETCH FICHIER
// ============================================================
async function fetchRemoteFile(fichier) {
  const url = BASE_URL + encodeURIComponent(fichier) + '?t=' + Date.now();
  const res = await fetch(url, { cache: 'no-store' });
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return await res.text();
}

// ============================================================
// CHARGEMENT + VÉRIFICATION MISE À JOUR
// ============================================================
async function loadAndCacheFile(fichier, forceRefresh = false) {
  if (!forceRefresh && cachedFiles[fichier]) {
    return cachedFiles[fichier];
  }

  const cached = await loadFromDB(fichier);
  let currentContent = cached ? cached.content : null;
  let currentVersion = cached ? cached.version : null;

  if (navigator.onLine) {
    try {
      const remoteVersion = await getRemoteVersion(fichier);
      const needUpdate = !currentContent || (remoteVersion && remoteVersion !== currentVersion);

      if (needUpdate) {
        const newContent = await fetchRemoteFile(fichier);
        if (!newContent || newContent.length < 10) {
          throw new Error("Fichier vide ou invalide");
        }
        const finalVersion = remoteVersion || new Date().toLocaleString('fr-FR');
        await saveToDB(fichier, newContent, finalVersion);
        currentContent = newContent;
        currentVersion = finalVersion;
        showToast('✅ ' + fichier + ' mis à jour');
      }
    } catch (e) {
      console.error(e);
      if (!currentContent) {
        throw new Error("Impossible de charger le contenu");
      }
    }
  } else {
    if (!currentContent) {
      throw new Error("Pas de connexion et pas de cache");
    }
  }

  cachedFiles[fichier] = { content: currentContent, version: currentVersion };
  return cachedFiles[fichier];
}

async function verificationAuto() {

      const cached = await loadFromDB(t.fichier);
      const localVersion = cached ? cached.version : null;
      
      // Raha niova ny version → maka fichier vaovao
      if (remoteVersion && remoteVersion !== localVersion) {
        console.log(`🆕 Mise à jour: ${t.fichier} (${localVersion} → ${remoteVersion})`);
        const newContent = await fetchRemoteFile(t.fichier);
        if (newContent && newContent.length > 10) {
          await saveToDB(t.fichier, newContent, remoteVersion);
          cachedFiles[t.fichier] = { content: newContent, version: remoteVersion };
          updateCount++;
        }
      }
    } catch (e) {
      console.warn(`⚠️ Erreur ${t.fichier}:`, e.message);
    }
  }
  
  if (updateCount > 0) {
    console.log(`✅ ${updateCount} fichier(s) mis à jour`);
    showToast(`✅ ${updateCount} fichier(s) mis à jour`);
  } else {
    console.log("✅ Tous les fichiers sont à jour");
  }
}

function demarrerVerificationAuto() {
  // 1. Vérification immédiate (après 3 secondes)
  setTimeout(() => verificationAuto(), 3000);
  
  // 2. Vérification toutes les 5 minutes
  if (autoUpdateInterval) clearInterval(autoUpdateInterval);
  autoUpdateInterval = setInterval(() => verificationAuto(), 5 * 60 * 1000);
  
  // 3. Vérification quand on revient sur l'onglet
  document.addEventListener('visibilitychange', () => {
    if (!document.hidden) verificationAuto();
  });
  
  // 4. Vérification quand on revient en ligne
  window.addEventListener('online', () => verificationAuto());
}

// PARSER LE FICHIER (structure libre)
// ============================================================
function parseFile(rawText) {
  const lines = rawText.split(/\r?\n/);
  const blocks = [];
  let currentBlock = { titre: "", description: "", contenu: [] };

  for (let i = 0; i < lines.length; i++) {
    const trimmed = lines[i].trim();

    const titleMatch = trimmed.match(/^Titre\s*:\s*(.+)$/i);
    if (titleMatch) {
      if (currentBlock.titre || currentBlock.contenu.length > 0) {
        blocks.push(currentBlock);
        currentBlock = { titre: "", description: "", contenu: [] };
      }
      currentBlock.titre = titleMatch[1].trim();
      continue;
    }

    const descMatch = trimmed.match(/^Description\s*:\s*(.+)$/i);
    if (descMatch) {
      currentBlock.description = descMatch[1].trim();
      continue;
    }

    if (trimmed) {
      currentBlock.contenu.push(trimmed);
    }
  }
  if (currentBlock.titre || currentBlock.contenu.length > 0) {
    blocks.push(currentBlock);
  }

  return blocks;
}

function filterTemps() {
  if (!currentSearch.trim()) return TEMPS_LISTE;
  const term = currentSearch.trim();

  if (/^\d+$/.test(term)) {
    const num = parseInt(term);
    return TEMPS_LISTE.filter(t => t.num === num);
  }

  return TEMPS_LISTE.filter(t =>
    fuzzyMatch(t.titre, term) ||
    fuzzyMatch(t.desc, term)
  );
}

    document.getElementById('backBtn').addEventListener('click', renderList);
  }
}

// AFFICHER LE CONTENU — HTML ao anaty iframe (CSS feno)
// ============================================================
function renderContent(tempsInfo, rawContent) {
  const isHTML = /<[a-z][\s\S]*>/i.test(rawContent);

  if (isHTML) {
    try {
      const parser = new DOMParser();
      const doc = parser.parseFromString(rawContent, "text/html");
      const body = doc.body;
      const head = doc.head;

      body.querySelectorAll("script").forEach(el => el.remove());
      head.querySelectorAll("script").forEach(el => el.remove());

      body.querySelectorAll("*").forEach(el => {
        Array.from(el.attributes).forEach(attr => {
          if (/^on/i.test(attr.name)) el.removeAttribute(attr.name);
        });
      });

      const fullHTML = `<!DOCTYPE html>
<html lang="fr">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  ${head.innerHTML}
  <style>
    body { font-family: 'Segoe UI', Roboto, sans-serif; padding: 16px; color: #2c3e50; background: #fffef7; line-height: 1.7; }
    h1 { color: #1e40af; border-bottom: 2px solid #c56985; padding-bottom: 6px; }
    h2 { color: #1e40af; }
    h3 { color: #c56985; }
    table { width: 100%; border-collapse: collapse; margin: 12px 0; }
    th, td { border: 1px solid #e2e8f0; padding: 8px 12px; text-align: left; }
    th { background: #eff6ff; color: #1e40af; }
    tr:nth-child(even) { background: #f8fafc; }
    code { background: #f1f5f9; padding: 2px 6px; border-radius: 4px; color: #b45309; }
    blockquote { border-left: 4px solid #c56985; padding-left: 16px; color: #64748b; font-style: italic; }
    img { max-width: 100%; height: auto; border-radius: 8px; }
    a { color: #1e40af; }
  </style>
</head>
<body>
  ${body.innerHTML}
</body>
</html>`;

      document.getElementById('app-root').innerHTML = `
        <div class="content-fullscreen">
          <div class="back-header">
            <button class="back-button" id="backBtn">← Retour</button>
          </div>
          <div class="content-header">
            <div class="content-titre">📖 ${escapeHtml(tempsInfo.titre)}</div>
            <div class="content-meta">${escapeHtml(tempsInfo.desc)}</div>
          </div>
          <div class="content-area" style="padding: 0; max-width: 100%;">
            <iframe 
              id="leconIframe" 
              class="content-iframe"
              sandbox="allow-same-origin allow-popups allow-popups-to-escape-sandbox"
            ></iframe>
          </div>
          <div class="content-footer">
            📚 Temps n°${tempsInfo.num} · Fahaizana
          </div>
        </div>
      `;

      const iframe = document.getElementById('leconIframe');
      iframe.srcdoc = fullHTML;

      iframe.onload = () => {
        try {
          const iframeDoc = iframe.contentDocument || iframe.contentWindow.document;
          const height = iframeDoc.body.scrollHeight;
          iframe.style.height = (height + 40) + 'px';
        } catch (e) {}
      };

      document.getElementById('backBtn').addEventListener('click', renderList);

    } catch (e) {
      console.error("iframe error:", e);
      document.getElementById('app-root').innerHTML = `
        <div class="content-fullscreen">
          <div class="back-header">
            <button class="back-button" id="backBtn">← Retour</button>
          </div>
          <div style="padding:2rem; text-align:center;">
            <div style="font-size:3rem;">⚠️</div>
            <h3 style="color:#c56985; margin:1rem 0;">Erreur HTML</h3>
            <p style="color:#4a2e38;">${escapeHtml(e.message)}</p>
            <button onclick="location.reload()" style="background:#c56985; color:white; border:none; padding:10px 24px; border-radius:40px; margin-top:1rem; cursor:pointer; font-weight:600;">
              🔄 Andramo indray
            </button>
          </div>
        </div>
      `;
      document.getElementById('backBtn').addEventListener('click', renderList);
    }
  } else {
    const blocks = parseFile(rawContent);
    const blocksHtml = blocks.map(b => `
      <div class="content-block">
        ${b.titre ? `<h3>📌 ${escapeHtml(b.titre)}</h3>` : ''}
        ${b.description ? `<p><em>${escapeHtml(b.description)}</em></p>` : ''}
        ${b.contenu.map(line => `<p>${escapeHtml(line)}</p>`).join('')}
      </div>
    `).join('');

    document.getElementById('app-root').innerHTML = `
      <div class="content-fullscreen">
        <div class="back-header">
          <button class="back-button" id="backBtn">← Retour</button>
        </div>
        <div class="content-header">
          <div class="content-titre">📖 ${escapeHtml(tempsInfo.titre)}</div>
          <div class="content-meta">${escapeHtml(tempsInfo.desc)}</div>
        </div>
        <div class="content-area">
          ${blocksHtml || '<p style="text-align:center; padding:2rem;">Fichier vide</p>'}
        </div>
        <div class="content-footer">
          📚 Temps n°${tempsInfo.num} · Fahaizana
        </div>
      </div>
    `;

    document.getElementById('backBtn').addEventListener('click', renderList);
  }
}
