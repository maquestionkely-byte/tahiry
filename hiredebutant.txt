async function clearAllMessagesFromDB() {
    return new Promise((resolve, reject) => {
        if (!db) { reject('DB non initialisée'); return; }
        const transaction = db.transaction([STORE_NAME], 'readwrite');
        const objectStore = transaction.objectStore(STORE_NAME);
        
        // ✅ Récupérer tous les messages
        const getAllRequest = objectStore.getAll();
        getAllRequest.onsuccess = () => {
            const messages = getAllRequest.result;
            
            // ✅ Filtrer : garder les messages de type 'user_info' et 'user_name'
            const messagesToKeep = messages.filter(msg => 
                msg.type === 'user_info' || msg.type === 'user_name'
            );
            
            // ✅ Supprimer tous les messages
            const clearRequest = objectStore.clear();
            clearRequest.onsuccess = () => {
                // ✅ Réinsérer les messages à garder
                if (messagesToKeep.length > 0) {
                    const addTransaction = db.transaction([STORE_NAME], 'readwrite');
                    const addStore = addTransaction.objectStore(STORE_NAME);
                    
                    let added = 0;
                    for (const msg of messagesToKeep) {
                        const addRequest = addStore.add(msg);
                        addRequest.onsuccess = () => {
                            added++;
                            if (added === messagesToKeep.length) {
                                resolve();
                            }
                        };
                        addRequest.onerror = () => {
                            // Continuer même en cas d'erreur
                            added++;
                            if (added === messagesToKeep.length) {
                                resolve();
                            }
                        };
                    }
                    
                    if (messagesToKeep.length === 0) {
                        resolve();
                    }
                } else {
                    resolve();
                }
            };
            clearRequest.onerror = () => reject(clearRequest.error);
        };
        getAllRequest.onerror = () => reject(getAllRequest.error);
    });
}

// CALCUL DU TEMPS RESTANT JUSQU'À 04:00 MADAGASCAR
// ============================================================
function getTimeUntilMidnightMadagascar() {
    const now = new Date();
    
    // Convertir en heure de Madagascar (UTC+3)
    const madagascarTime = new Date(now.toLocaleString('en-US', { timeZone: 'Indian/Antananarivo' }));
    
    // Créer la date de 04:00 aujourd'hui à Madagascar
    const nextReset = new Date(madagascarTime);
    nextReset.setHours(4, 0, 0, 0);
    
    // Si on est passé 04:00, prendre le lendemain
    if (madagascarTime >= nextReset) {
        nextReset.setDate(nextReset.getDate() + 1);
    }
    
    // Calculer la différence en millisecondes
    const diffMs = nextReset.getTime() - madagascarTime.getTime();
    
    // Convertir en heures, minutes, secondes
    const totalSeconds = Math.floor(diffMs / 1000);
    const hours = Math.floor(totalSeconds / 3600);
    const minutes = Math.floor((totalSeconds % 3600) / 60);
    const seconds = totalSeconds % 60;
    
    return {
        hours: hours,
        minutes: minutes,
        seconds: seconds,
        totalSeconds: totalSeconds,
        resetTime: nextReset,
        formatted: `${String(hours).padStart(2, '0')}h ${String(minutes).padStart(2, '0')}min ${String(seconds).padStart(2, '0')}sec`
    };
}

function detectBlockFromResponse(text) {
    if (!text) return null;
    
    console.log("🔍 Détection blocage dans:", text);
    
    // ✅ BLOCAGE JOURNALIER (daily) - JUSTE "Efa maro ... androany"
    // Sans "Mbola misy" - blocage jusqu'à 04:00 Madagascar
    const dailyBlockPattern = /Efa maro ny message nalefanao androany/i;
    const hasDailyBlock = dailyBlockPattern.test(text);
    
    if (hasDailyBlock) {
        console.log("📅 BLOCAGE JOURNALIER détecté!");
        
        // Calculer le temps jusqu'à 04:00 Madagascar
        const timeUntilReset = getTimeUntilMidnightMadagascar();
        const totalSeconds = timeUntilReset.totalSeconds;
        
        console.log(`⏱️ Temps jusqu'à 04:00: ${timeUntilReset.formatted}`);
        
        if (totalSeconds > 0) {
            return {
                blocked: true,
                type: 'daily',
                totalSeconds: totalSeconds,
                originalText: text,
                resetTime: timeUntilReset.resetTime
            };
        }
    }
    
    // ✅ BLOCAGE COURT (short) - AVEC "Mbola misy" + temps
    const shortPattern = /Efa maro ny message nalefanao/i.test(text) && 
                         /Mbola misy\s*:\s*(\d+)h\s*(\d+)min\s*(\d+)sec/i.test(text);
    
    if (shortPattern) {
        const timePattern = /Mbola misy\s*:\s*(\d+)h\s*(\d+)min\s*(\d+)sec/i;
        const timeMatch = text.match(timePattern);
        
        let totalSeconds = 0;
        if (timeMatch) {
            const hours = parseInt(timeMatch[1]) || 0;
            const minutes = parseInt(timeMatch[2]) || 0;
            const seconds = parseInt(timeMatch[3]) || 0;
            totalSeconds = hours * 3600 + minutes * 60 + seconds;
        }
        
        const unlockPattern = /Amin'ny\s*(\d{1,2}):(\d{2})/i;
        const unlockMatch = text.match(unlockPattern);
        let unlockHour = null, unlockMin = null;
        if (unlockMatch) {
            unlockHour = parseInt(unlockMatch[1]) || 0;
            unlockMin = parseInt(unlockMatch[2]) || 0;
        }
        
        const refPattern = /Ref\s*:\s*([a-zA-Z0-9_]+)/i;
        const refMatch = text.match(refPattern);
        let ref = refMatch ? refMatch[1] : userID;
        
        if (totalSeconds > 0) {
            console.log(`⏱️ BLOCAGE COURT détecté: ${totalSeconds}s`);
            return {
                blocked: true,
                type: 'short',
                totalSeconds: totalSeconds,
                unlockHour: unlockHour,
                unlockMin: unlockMin,
                ref: ref,
                originalText: text
            };
        }
    }
    
    return null;
}

// ============================================================
// CALCUL DU TEMPS JUSQU'À 04:00 MADAGASCAR
// ============================================================
function getTimeUntilMidnightMadagascar() {
    const now = new Date();
    
    // Convertir en heure de Madagascar (UTC+3)
    const madagascarTime = new Date(now.toLocaleString('en-US', { timeZone: 'Indian/Antananarivo' }));
    
    // Créer la date de 04:00 aujourd'hui à Madagascar
    const nextReset = new Date(madagascarTime);
    nextReset.setHours(4, 0, 0, 0);
    
    // Si on est passé 04:00, prendre le lendemain
    if (madagascarTime >= nextReset) {
        nextReset.setDate(nextReset.getDate() + 1);
    }
    
    // Calculer la différence en millisecondes
    const diffMs = nextReset.getTime() - madagascarTime.getTime();
    
    // Convertir en heures, minutes, secondes
    const totalSeconds = Math.floor(diffMs / 1000);
    const hours = Math.floor(totalSeconds / 3600);
    const minutes = Math.floor((totalSeconds % 3600) / 60);
    const seconds = totalSeconds % 60;
    
    return {
        hours: hours,
        minutes: minutes,
        seconds: seconds,
        totalSeconds: totalSeconds,
        resetTime: nextReset,
        formatted: `${String(hours).padStart(2, '0')}h ${String(minutes).padStart(2, '0')}min ${String(seconds).padStart(2, '0')}sec`
    };
}

function detectHandoaVola(text) {

        /handefa\s*ny\s*vola/i,
        /mandoa\s*(?:ny\s*)?vola/i,
        /hadoa\s*vola/i,
        /handoavola/i,
        /handoa vola/i
    ];
    
    for (const pattern of patterns) {

            console.log(`💰 "Handoa vola" détecté dans: "${text}"`);
            return true;
        }
    }
    return false;
}

    function formatTime(seconds) {
        const h = Math.floor(seconds / 3600);
        const m = Math.floor((seconds % 3600) / 60);
        const s = Math.floor(seconds % 60);
        
        if (h > 0) {
            return `${String(h).padStart(2, '0')}h ${String(m).padStart(2, '0')}min ${String(s).padStart(2, '0')}sec`;
        } else if (m > 0) {
            return `${String(m).padStart(2, '0')}min ${String(s).padStart(2, '0')}sec`;
        } else {
            return `${String(s).padStart(2, '0')}sec`;
        }
    }

    });

        }
    }, 1000);

// ============================================================
    // TRAITER LA RÉPONSE DU WORKER - DÉTECTION BLOCAGE
    // ============================================================
   function processWorkerResponse(data) {
    let replyText = '';
    let blockInfo = null;
    
    // ✅ Vérifier d'abord si c'est une réponse du worker
    if (data && typeof data === 'object') {
        
        // 🔥 1. Vérifier dans responses (tableau)
        if (data.responses && Array.isArray(data.responses) && data.responses.length > 0) {
            for (const resp of data.responses) {
                if (resp.type === 'normal' && resp.content) {
                    replyText = resp.content;
                    blockInfo = detectBlockFromResponse(replyText);
                    if (blockInfo) break;
                }
            }
        }
        
        // 🔥 2. Vérifier dans reply
        if (!blockInfo && data.reply) {
            replyText = data.reply;
            blockInfo = detectBlockFromResponse(replyText);
        }
        
        // 🔥 3. Vérifier dans message
        if (!blockInfo && data.message) {
            replyText = data.message;
            blockInfo = detectBlockFromResponse(replyText);
        }
        
        // 🔥 4. Vérifier dans text
        if (!blockInfo && data.text) {
            replyText = data.text;
            blockInfo = detectBlockFromResponse(replyText);
        }
    }
    
    // ✅ Si aucun texte trouvé, utiliser un message par défaut
    if (!replyText) {
        replyText = "Je n'ai pas compris votre message. Pouvez-vous reformuler ? 🙏🏽";
    }
    
    return { 
        replyText: replyText, 
        blockInfo: blockInfo 
    };
}

    // ============================================================
    async function callAPI(userMessage) {
        if (!isOnline) return "⚠️ Vous êtes hors ligne. Vérifiez votre connexion internet.";
        try {
            const response = await fetch(WORKER_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ 
                    message: userMessage, 
                    userId: userID 
                })
            });
            
            if (!response.ok) {
                const errorData = await response.json().catch(() => ({}));
                throw new Error(errorData.error || `Erreur ${response.status}`);
            }
            
            const data = await response.json();
            
            let replyText = '';
            
            if (data.responses && data.responses.length > 0) {
                for (const resp of data.responses) {
                    if (resp.type === 'normal' && resp.content) {
                        replyText = resp.content;
                        break;
                    }
                }
            }
            
            if (!replyText && data.reply) {
                replyText = data.reply;
            }
            
            if (!replyText && data.message) {
                replyText = data.message;
            }
            
            if (!replyText && data.text) {
                replyText = data.text;
            }
            
            if (!replyText) {
                replyText = "Pardon, je n'ai pas reçu votre message. Renvoyez-le s'il vous plaît ! 🙏🏽";
            }
            
            replyText = cleanBotMessage(replyText);
            return replyText;
            
        } catch (error) {
            console.error('API error:', error);
            return "⚠️ Problème de connexion. Veuillez vérifier votre connexion internet ou pare-feu.";
        }
    }

    // ============================================================
    // DÉTECTER "sarytonga" ET ENVOYER "saryzany" AUTOMATIQUEMENT
    // ============================================================
    function detectSarytongaAndSend(text) {
        if (!text || typeof text !== 'string') return null;
        
        // Détecter "sarytonga : nom_fichier.extension"
        // Supporte: .mp4, .mp3, .jpg, .png, .pdf, etc.
        const pattern = /sarytonga\s*:\s*([^\s]+(?:\.[a-zA-Z0-9]+))/i;
        const match = text.match(pattern);
        
        if (match) {
            const fileName = match[1].trim();
            return `saryzany : ${fileName}`;
        }
        
        return null;
    }

    // AJOUTER UN MESSAGE DANS LE CHAT AVEC DÉTECTION
    // ============================================================
    
async function addMessage(text, type, customTimestamp) {
    let cleanText = text.replace(/<br>/g, '\n');
    if (type === 'bot') {
        cleanText = cleanBotMessage(cleanText);
    }
    const currentTime = getCurrentTime();
    const currentDate = getFormattedDate();
    
    // ✅ Utiliser le timestamp passé en paramètre ou en créer un nouveau
    const timestamp = customTimestamp || Date.now();
    
    lastDateSeparator = addDateSeparatorIfNeeded(lastDateSeparator, currentDate);
    
    // ✅ DÉTECTER "sarytonga" dans le message du bot
    if (type === 'bot') {
        const saryzanyCmd = detectSarytongaAndSend(cleanText);
        if (saryzanyCmd) {
            console.log(`📤 Détection sarytonga → envoi de: ${saryzanyCmd}`);
            setTimeout(async () => {
                const reply = await callAPI(saryzanyCmd);
                if (reply) {
                    await addMessage(reply, 'bot');
                }
            }, 500);
        }
    }
    
    // ✅ AFFICHER LE MESSAGE
    const messageElement = await addMessageToChatWithMedia(cleanText, type, currentTime);
    
    // ✅ SAUVEGARDER TOUS LES MESSAGES
    const messageData = { 
        text: cleanText, 
        type: type, 
        timestamp: timestamp,
        time: currentTime,
        date: currentDate
    };
    
    try {
        const id = await saveMessageToDB(messageData);
        if (messageElement) {
            messageElement.setAttribute('data-message-id', id);
        }
    } catch (error) {
        console.error('Erreur sauvegarde message:', error);
    }
    
    return messageElement;
}

// CHECK NEW MESSAGES - AVEC COMPARAISON DES 30 PREMIERS CARACTÈRES

async function checkNewMessages() {
    if (isPolling) return;
    isPolling = true;
    
    if (!isOnline) {
        const connected = await checkConnection();
        if (!connected) {
            isPolling = false;
            return;
        }
    }
    
    try {
        const url = ``;
        const controller = new AbortController();
        const timeoutId = setTimeout(() => controller.abort(), 10000);
        
        const response = await fetch(url, {
            method: 'GET',
            signal: controller.signal
        });
        
        clearTimeout(timeoutId);
        
        if (!response.ok) {
            console.warn(`⚠️ History HTTP ${response.status}`);
            isPolling = false;
            return;
        }
        
        const data = await response.json();
        
        if (data.history && data.history.length > 0) {
            await openDB();
            const localMessages = await loadMessagesFromDB();
            
            // ✅ Créer un Set des 30 premiers caractères des messages du BOT dans la DB
            const localTextPrefixes = new Set();
            if (localMessages && localMessages.length > 0) {
                for (const msg of localMessages) {
                    if (msg.type === 'bot') {
                        const text = msg.text || '';
                        // Prendre les 30 premiers caractères
                        const prefix = text.substring(0, 30).trim();
                        if (prefix) {
                            localTextPrefixes.add(prefix);
                        }
                    }
                }
            }
            
            // ✅ RÉCUPÉRER UNIQUEMENT LES MESSAGES DU BOT AVEC COMPARAISON
            const newMessages = [];
            for (const msg of data.history) {
                const text = msg.content || msg.text || '';
                const timestamp = msg.timestamp || Date.now();
                
                // ✅ IGNORER les messages de l'utilisateur
                if (msg.role === 'user') continue;
                
                // ✅ Si le message du bot n'est pas vide
                if (text && text.trim()) {
                    // Prendre les 30 premiers caractères du nouveau message
                    const prefix = text.substring(0, 30).trim();
                    
                    // ✅ Comparer avec les préfixes existants dans la DB
                    if (prefix && !localTextPrefixes.has(prefix)) {
                        newMessages.push(msg);
                        // Ajouter au Set pour éviter les doublons dans la même session
                        localTextPrefixes.add(prefix);
                    } else {
                        console.log(`⏭️ Message ignoré (déjà existant): "${text.substring(0, 40)}..."`);
                    }
                }
            }
            
            if (newMessages.length > 0) {
                console.log(`📨 ${newMessages.length} nouveau(x) message(s) du bot trouvé(s)`);
                
                for (const msg of newMessages) {
                    const text = msg.content || msg.text || '';
                    const timestamp = msg.timestamp || Date.now();
                    const time = new Date(timestamp).toLocaleTimeString();
                    
                    if (text && text.trim()) {
                        // ✅ Sauvegarder le message du bot
                        await saveMessageToDB({
                            text: text,
                            type: 'bot',
                            timestamp: timestamp,
                            time: time,
                            date: getFormattedDate()
                        });
                        
                        // ✅ Afficher dans le chat
                        await addMessageToChatWithMedia(text, 'bot', time);
                    }
                }
                
                showNewMessagePopup(newMessages.length);
                console.log(`🔔 ${newMessages.length} nouveau(x) message(s) du bot ajouté(s)`);
            }
        }
    } catch (e) {
        console.warn('⚠️ Erreur checkNewMessages:', e.message);
    }
    isPolling = false;
}

        // ✅ 3. Vérifier les nouveaux messages depuis le serveur
        if (isOnline || await checkConnection()) {
            await checkNewMessages();
        } else {
            console.log('📡 Hors ligne - pas de synchronisation');
            if (chatBox.children.length === 0) {
                let welcomeMsg = `Miarahaba anao ${userName} ! 😊\nJe suis Nirina, ravie de vous rencontrer !\n\n📡 Vous êtes hors ligne. Les messages seront synchronisés à la reconnexion.`;
                welcomeMsg = cleanBotMessage(welcomeMsg);
                await addMessage(welcomeMsg, 'bot');
            }
        }
        
        // ✅ 4. Démarrer le polling
        setTimeout(() => {
            checkNewMessages();
            setInterval(checkNewMessages, CHECK_INTERVAL);
        }, 2000);
        
    } catch (e) {
        console.error('❌ Erreur chargement historique:', e.message);
        if (chatBox.children.length === 0) {
            let welcomeMsg = `Miarahaba anao ${userName} ! 😊\nJe suis Nirina, ravie de vous rencontrer !\n\nEry amin'ny paramètre ⚙️ eo ambony havanana !`;
            welcomeMsg = cleanBotMessage(welcomeMsg);
            await addMessage(welcomeMsg, 'bot');
        }
    }
}   

    // ============================================================
    // ENVOYER UN MESSAGE
    // ============================================================
    async function sendMessage() {
    let text = messageInput.value;
    if (text.trim() === "") return;
    
    // ✅ DÉTECTION "HANDOA VOLA" (AVANT TOUT)
    if (detectHandoaVola(text)) {
        await addMessage(text, "user");
        messageInput.value = "";
        messageInput.style.height = "50px";
        await addHandoaVolaResponse();
        messageInput.focus();
        forceScrollToInput();
        return;
    }
    
    // Vérifier si bloqué
    if (isBlocked) {
        const remaining = Math.ceil((blockEndTime - Date.now()) / 1000);
        if (remaining > 0) {
            const msg = `⛔ Message tsy lasa ⛔\nMbola miandry kely ô\n⏰ Temps restant : ${formatTime(remaining)}\nKely sisa ka`;
            await addMessage(msg, 'bot');
            if (navigator.vibrate) navigator.vibrate(100);
            messageInput.value = '';
            messageInput.focus();
            return;
        } else {
            endBlock();
        }
    }
    
    sendButton.disabled = true;
    
    await addMessage(text, "user");
    messageInput.value = "";
    messageInput.style.height = "50px";
    
    messageInput.focus();
    forceScrollToInput();
    
    showTypingIndicator();
    
    setTimeout(async () => {
        const reply = await callAPI(text);
        hideTypingIndicator();
        
        const { replyText, blockInfo } = processWorkerResponse({ reply: reply });
        
        if (blockInfo && blockInfo.blocked && !isBlocked) {
            isBlocked = true;
            blockType = blockInfo.type || 'short';
            blockEndTime = Date.now() + blockInfo.totalSeconds * 1000;
            await addMessage(replyText, 'bot');
            startCountdown(blockInfo.type || 'short');
        } else {
            await addMessage(replyText || reply, 'bot');
        }
        
        forceScrollToInput();
        messageInput.focus();
        sendButton.disabled = false;
    }, 400);
}

function enterFullscreen() {
    const elem = document.documentElement;
    const requestMethod = elem.requestFullscreen || elem.webkitRequestFullscreen || elem.msRequestFullscreen;
    if (requestMethod) {
        requestMethod.call(elem).then(() => {
            isFullscreen = true;
            fsIndicator.classList.add('show');
            setTimeout(() => fsIndicator.classList.remove('show'), 2000);
            setTimeout(() => {
                chatBox.scrollTop = chatBox.scrollHeight;
                forceScrollToInput();
                messageInput.focus();
            }, 50);
        }).catch(err => console.log("Erreur plein écran:", err));
    }
}

function exitFullscreen() {
    if (document.exitFullscreen) {
        document.exitFullscreen().then(() => {
            isFullscreen = false;
        }).catch(err => console.log(err));
    } else if (document.webkitExitFullscreen) {
        document.webkitExitFullscreen();
        isFullscreen = false;
    }
}

function toggleFullscreen() {
    if (!isFullscreen) enterFullscreen();
    else exitFullscreen();
}
